' error' : ''); requestAnimationFrame(() => { t.classList.add('show'); }); setTimeout(() => t.classList.remove('show'), 3000); } // ── Mark active nav link ───────────────────────────────────────────────────── function setActiveNav() { const page = location.pathname.split('/').pop() || 'index.html'; document.querySelectorAll('nav a').forEach(a => { const href = a.getAttribute('href'); if (href === page || (page === 'index.html' && href === 'index.html')) { a.classList.add('active'); } }); } // ── Format date for display ────────────────────────────────────────────────── function fmtDate(d) { if (!d) return '—'; return new Date(d + 'T12:00:00').toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' }); } // ── Format currency ────────────────────────────────────────────────────────── function fmtMoney(n) { if (n == null) return '—'; return '$' + Number(n).toFixed(2); } // ── Today's date as YYYY-MM-DD ─────────────────────────────────────────────── function today() { return new Date().toISOString().split('T')[0]; } // ── Run on every page ──────────────────────────────────────────────────────── document.addEventListener('DOMContentLoaded', setActiveNav);
